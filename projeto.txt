  function calculatePriority(dueDateStr, weight, maxScore = 10) {
    const today = new Date('2026-11-09');
    const due = new Date(dueDateStr + 'T00:00:00');
    const daysLeft = Math.ceil((due - today) / (1000 * 60 * 60 * 24));
    const mathIndex = ((maxScore * (weight === 'Alto' ? 1.5 : 1.0)) / Math.max(1, daysLeft + 1)).toFixed(1);
    let deadlineLabel = `Em ${daysLeft} dias`;
    if (daysLeft < 0) deadlineLabel = `Atrasada`;
    else if (daysLeft === 0) deadlineLabel = 'Vence Hoje!';
    
    if (daysLeft <= 2 || (daysLeft <= 5 && weight === 'Alto')) {
      return { level: 'Crítica', rank: 1, mathIndex, deadlineLabel, badgeClass: 'bg-rose-100 text-rose-800 border-rose-300', borderClass: 'border-l-[6px] border-l-rose-500' };
    }
    if (daysLeft <= 7) {
      return { level: 'Moderada', rank: 2, mathIndex, deadlineLabel, badgeClass: 'bg-amber-100 text-amber-800 border-amber-300', borderClass: 'border-l-[6px] border-l-amber-500' };
    }
    return { level: 'Planejada', rank: 3, mathIndex, deadlineLabel, badgeClass: 'bg-emerald-100 text-emerald-800 border-emerald-300', borderClass: 'border-l-[6px] border-l-emerald-500' };
  }

  function setFilter(filter) { currentFilter = filter; renderTasks(); }

  function getVisibleTasks() {
    return tasks.map(t => ({ ...t, priority: calculatePriority(t.dueDate, t.weight, t.maxScore || 10) }));
  }

  function renderTasks() {
    const container = document.getElementById('tasks-container');
    container.innerHTML = '';
    const visibleTasks = getVisibleTasks();
    let pending = 0, critical = 0, pomos = 0;

    visibleTasks.forEach(t => {
      const md = getMyTaskData(t.id);
      pomos += md.pomodoros || 0;
      if (!md.completed) {
        pending++;
        if (t.priority.level === 'Crítica') critical++;
      }
    });

    document.getElementById('stat-pending').textContent = pending;
    document.getElementById('stat-critical').textContent = critical;
    document.getElementById('stat-pomodoros').textContent = pomos;
    updateOverallAverageDisplay();

    const filtered = visibleTasks.filter(t => {
      const md = getMyTaskData(t.id);
      if (currentFilter === 'done') return md.completed;
      if (md.completed) return false;
      if (currentFilter === 'critical') return t.priority.level === 'Crítica';
      if (currentFilter === 'turma') return t.scope === 'turma';
      return true;
    });

    filtered.forEach(task => {
      const md = getMyTaskData(task.id);
      const subTotal = md.subtasks.length;
      const subDone = md.subtasks.filter(s => s.done).length;
      const pct = md.completed ? 100 : (subTotal === 0 ? 0 : Math.round((subDone / subTotal) * 100));

      const subHtml = md.subtasks.map((sub, idx) => `
        <label class="flex items-center justify-between gap-2 p-2 rounded-xl bg-slate-50 cursor-pointer">
          <div class="flex items-center gap-2">
            <input type="checkbox" ${sub.done ? 'checked' : ''} onchange="toggleSub('${task.id}', ${idx})" class="w-4 h-4 text-orange-600">
            <span class="text-xs ${sub.done ? 'line-through text-slate-400' : 'text-slate-700'}">${sub.text}</span>
          </div>
          <button onclick="remSub('${task.id}', ${idx})" class="text-slate-300 hover:text-rose-500"><i data-lucide="trash-2" class="w-3 h-3"></i></button>
        </label>
      `).join('');

      const btnSteps = (subTotal === 0 && !md.completed) ? `
        <button onclick="suggestSteps('${task.id}')" class="w-full mt-1.5 py-2 px-3 rounded-2xl bg-orange-50 text-orange-700 text-xs font-extrabold flex items-center justify-center gap-1.5">
          <i data-lucide="sparkles" class="w-4 h-4 text-amber-500"></i> Sugerir passos iniciais (RN05)
        </button>
      ` : '';

      const card = document.createElement('div');
      card.className = `bg-white rounded-card p-4 shadow-sm border border-slate-200 ${task.priority.borderClass} space-y-2`;
      card.innerHTML = `
        <div class="flex items-center justify-between">
          <span class="px-2.5 py-0.5 text-[10px] font-extrabold uppercase rounded-full border ${task.priority.badgeClass}">${task.priority.level} • Índice ${task.priority.mathIndex}</span>
          <button onclick="toggleComp('${task.id}')" class="text-[11px] font-extrabold px-2.5 py-1 rounded-full ${md.completed ? 'bg-amber-100 text-amber-800' : 'bg-emerald-100 text-emerald-800'}">${md.completed ? 'Reabrir' : '✔ Concluir'}</button>
        </div>
        <h4 class="text-sm font-extrabold ${md.completed ? 'line-through text-slate-400' : 'text-slate-900'}">${task.title}</h4>
        <div class="flex items-center justify-between text-xs text-slate-600 bg-slate-50 px-3 py-1.5 rounded-xl">
          <span>📅 ${task.priority.deadlineLabel}</span>
          <button onclick="startFocus('${task.id}')" class="text-orange-600 font-extrabold flex items-center gap-1"><i data-lucide="play-circle" class="w-4 h-4"></i> Focar (${md.pomodoros || 0})</button>
        </div>
        <div>
          <div class="flex justify-between text-[11px] font-bold text-slate-500 mb-1"><span>Progresso</span><span>${pct}%</span></div>
          <div class="w-full h-2 bg-slate-100 rounded-full overflow-hidden"><div class="h-full bg-orange-600" style="width: ${pct}%"></div></div>
        </div>
        <div class="space-y-1">${subHtml} ${btnSteps}</div>
        <form onsubmit="addSub(event, '${task.id}')" class="flex gap-1 pt-1">
          <input type="text" id="sub-${task.id}" placeholder="Nova microtarefa..." class="flex-1 px-3 py-1.5 text-xs rounded-xl border border-slate-200">
          <button type="submit" class="bg-slate-900 text-white text-xs font-extrabold px-3 py-1.5 rounded-xl">+</button>
        </form>
      `;
      container.appendChild(card);
    });
    lucide.createIcons();
  }

  function toggleComp(id) { getMyTaskData(id).completed = !getMyTaskData(id).completed; renderTasks(); }
  function suggestSteps(id) {
    const md = getMyTaskData(id);
    md.subtasks = [{ text: '1. Ler instruções e separar materiais', done: false }, { text: '2. Desenvolver rascunho', done: false }, { text: '3. Revisar e entregar', done: false }];
    renderTasks();
  }
  function addSub(e, id) {
    e.preventDefault();
    const inp = document.getElementById(`sub-${id}`);
    if (!inp.value.trim()) return;
    getMyTaskData(id).subtasks.push({ text: inp.value.trim(), done: false });
    inp.value = '';
    renderTasks();
  }
  function toggleSub(id, idx) { getMyTaskData(id).subtasks[idx].done = !getMyTaskData(id).subtasks[idx].done; renderTasks(); }
  function remSub(id, idx) { getMyTaskData(id).subtasks.splice(idx, 1); renderTasks(); }

  function populateSubjectSelect() {
    document.getElementById('task-subject').innerHTML = disciplines.map(d => `<option value="${d.name}">${d.name}</option>`).join('');
  }

  function openTaskModal() { document.getElementById('task-modal').classList.remove('hidden'); }
  function closeTaskModal() { document.getElementById('task-modal').classList.add('hidden'); }

  document.getElementById('task-form').addEventListener('submit', (e) => {
    e.preventDefault();
    tasks.push({
      id: 'task_' + Date.now(), classCode: currentUser.classCode, scope: document.getElementById('task-scope').value,
      title: document.getElementById('task-title').value, subject: document.getElementById('task-subject').value,
      weight: document.getElementById('task-weight').value, dueDate: document.getElementById('task-date').value,
      maxScore: parseFloat(document.getElementById('task-max-score').value) || 10, description: document.getElementById('task-desc').value
    });
    closeTaskModal();
    renderTasks();
    renderDisciplinesAndGrades();
    populatePomoSelect();
  });

  function addDiscipline(e) {
    e.preventDefault();
    const name = document.getElementById('new-disc-name').value.trim();
    if (!name) return;
    disciplines.push({ id: 'disc_' + Date.now(), name: name, color: document.getElementById('new-disc-color').value });
    document.getElementById('new-disc-name').value = '';
    populateSubjectSelect();
    renderDisciplinesAndGrades();
  }

  function saveGrade(id, max) {
    const val = parseFloat(document.getElementById(`grade-${id}`).value);
    if (isNaN(val) || val < 0 || val > max) { alert('Nota inválida (RN07)!'); return; }
    getMyTaskData(id).grade = val;
    renderDisciplinesAndGrades();
    updateOverallAverageDisplay();
  }

  function updateOverallAverageDisplay() {
    const grades = tasks.map(t => getMyTaskData(t.id).grade).filter(g => g !== null && g !== undefined);
    document.getElementById('stat-avg').textContent = grades.length === 0 ? '--' : (grades.reduce((a,b)=>a+b,0)/grades.length).toFixed(1);
  }

  function renderDisciplinesAndGrades() {
    document.getElementById('disciplines-container').innerHTML = disciplines.map(d => `
      <div class="flex items-center justify-between p-2 rounded-xl bg-slate-50 border border-slate-200">
        <span class="text-xs font-bold" style="color: ${d.color}">📚 ${d.name}</span>
      </div>
    `).join('');

    document.getElementById('grades-list-container').innerHTML = tasks.map(t => `
      <div class="flex items-center justify-between gap-2 p-2 rounded-xl bg-slate-50 border border-slate-200">
        <span class="text-xs font-bold truncate">${t.title}</span>
        <div class="flex items-center gap-1">
          <input type="number" id="grade-${t.id}" step="0.1" value="${getMyTaskData(t.id).grade ?? ''}" placeholder="Nota" class="w-16 px-2 py-1 text-xs text-center border rounded-xl">
          <button onclick="saveGrade('${t.id}', ${t.maxScore})" class="bg-emerald-600 text-white text-[10px] font-bold px-2 py-1 rounded-xl">Salvar</button>
        </div>
      </div>
    `).join('');
  }

  function switchTab(tab) {
    ['mural', 'foco', 'notas'].forEach(t => {
      document.getElementById(`tab-${t}`).classList.toggle('hidden', t !== tab);
      document.getElementById(`nav-${t}`).className = t === tab ? 'flex flex-col items-center gap-0.5 text-orange-600 bg-orange-50 font-extrabold text-[11px] py-1.5 px-4 rounded-2xl' : 'flex flex-col items-center gap-0.5 text-slate-400 font-bold text-[11px] py-1.5 px-4 rounded-2xl';
    });
    if (tab === 'foco') populatePomoSelect();
    if (tab === 'notas') renderDisciplinesAndGrades();
  }

  function populatePomoSelect() {
    document.getElementById('pomo-task-select').innerHTML = tasks.map(t => `<option value="${t.id}">${t.title}</option>`).join('');
    if (tasks.length > 0) selectedTaskIdForPomo = tasks[0].id;
  }
  function selectPomoTask(id) { selectedTaskIdForPomo = id; }
  function startFocus(id) { selectedTaskIdForPomo = id; switchTab('foco'); }

  function togglePomodoro() {
    if (pomoIsRunning) {
      clearInterval(pomoInterval);
      pomoIsRunning = false;
      document.getElementById('btn-pomo-toggle').textContent = 'Retomar';
    } else {
      pomoIsRunning = true;
      document.getElementById('btn-pomo-toggle').textContent = 'Pausar';
      pomoInterval = setInterval(() => {
        if (pomoSecondsLeft > 0) {
          pomoSecondsLeft--;
          let m = String(Math.floor(pomoSecondsLeft/60)).padStart(2,'0');
          let s = String(pomoSecondsLeft%60).padStart(2,'0');
          document.getElementById('pomo-timer-display').textContent = `${m}:${s}`;
        } else {
          clearInterval(pomoInterval);
          if (selectedTaskIdForPomo) { getMyTaskData(selectedTaskIdForPomo).pomodoros++; }
          alert('Ciclo Pomodoro concluído em 00:00! +1 bloco registrado (RN06).');
          pomoSecondsLeft = 25*60;
          pomoIsRunning = false;
          renderTasks();
        }
      }, 1000);
    }
  }
  function resetPomodoro() { clearInterval(pomoInterval); pomoIsRunning = false; pomoSecondsLeft = 25*60; document.getElementById('pomo-timer-display').textContent = '25:00'; }
  function startDemoPomodoro() { clearInterval(pomoInterval); pomoSecondsLeft = 5; togglePomodoro(); }
